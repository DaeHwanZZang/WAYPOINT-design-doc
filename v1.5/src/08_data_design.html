8. 데이터 설계

FMS는 현재 상태는 JSON 파일, 쌓이는 이력은 SQLite(events.db)에 둔다. 단일 프로세스·worker 1개 전제에서 파일 저장소 + 프로세스 내부 락이 가장 단순하고, 재시작 복구(NFR-05)를 파일 원자 교체로 보장할 수 있기 때문이다. 관계형 DB 서버(v1.0의 MySQL)는 쓰지 않는다. 모든 영속 데이터는 FMS_DATA_ROOT(기본 fms_server/maps/, git 제외) 아래에 있다.

SQLite 테이블 (물리 설계 — SQL 스크립트)

-- fms_server/events.py · alerts.py · webhooks.py (구현). PRAGMA journal_mode=WAL
CREATE TABLE IF NOT EXISTS events(
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ts REAL NOT NULL,                       -- unix time (초)
    site_id TEXT,
    kind TEXT NOT NULL,                     -- 8.3 이벤트 카탈로그
    severity TEXT NOT NULL DEFAULT 'info',  -- info | warning | error
    robot_id TEXT,
    job_id TEXT,
    actor TEXT,                             -- API 키 이름 / 'system'
    data TEXT NOT NULL DEFAULT '{}'         -- JSON
);
CREATE INDEX IF NOT EXISTS events_site_ts  ON events(site_id, ts);
CREATE INDEX IF NOT EXISTS events_robot_ts ON events(robot_id, ts);
CREATE INDEX IF NOT EXISTS events_job      ON events(job_id);
CREATE INDEX IF NOT EXISTS events_kind_ts  ON events(kind, ts);

CREATE TABLE IF NOT EXISTS alerts(
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    site_id TEXT, rule TEXT NOT NULL, key TEXT NOT NULL,   -- 같은 key 는 열린 알림 하나만
    severity TEXT NOT NULL, message TEXT NOT NULL, data TEXT NOT NULL DEFAULT '{}',
    opened_at REAL NOT NULL, acked_by TEXT, acked_at REAL, resolved_at REAL
);

CREATE TABLE IF NOT EXISTS webhook_queue(
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    webhook_id TEXT NOT NULL, event_id INTEGER, payload TEXT NOT NULL,
    attempts INTEGER NOT NULL DEFAULT 0, next_at REAL NOT NULL, created_at REAL NOT NULL, last_error TEXT
);
CREATE INDEX IF NOT EXISTS webhook_queue_due ON webhook_queue(next_at);

-- 이번 개발 (계획): 운행 단위 기록
ALTER TABLE events ADD COLUMN run_id TEXT;            -- 시작 시 컬럼 없으면 추가 (하위 호환)
CREATE INDEX IF NOT EXISTS events_run ON events(run_id, ts);
CREATE TABLE IF NOT EXISTS runs(
    run_id TEXT PRIMARY KEY, site_id TEXT, condition INTEGER, robots INTEGER, pattern TEXT,
    seed INTEGER, meta TEXT NOT NULL DEFAULT '{}', started_at REAL NOT NULL, stopped_at REAL
);
