실행 명령

# AMR-fms 리포 루트
python3 -m venv .venv && .venv/bin/pip install -r requirements.txt
.venv/bin/uvicorn fms_server.main:app --host 127.0.0.1 --port 8000      # UI: / , API 문서: /docs , 상태: /health
.venv/bin/python -m pytest fms_server/tests/ -q                         # 전체 (약 7분)
node --test fms_server/tests/zone-geometry.test.js

# AMR-simulator 리포 루트
make                                                                    # robot_client 빌드
mosquitto -c broker/mosquitto.conf -v
./build/bin/robot_client --id AMR-001 --host 127.0.0.1
.venv/bin/python tools/launcher.py                                      # 통합 런처 (브로커·로봇 N대·뷰어)
.venv/bin/python tools/run_experiment.py experiments/s10_crowded.yaml   # (계획) FMS 연동 실험 러너
